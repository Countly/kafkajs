const Encoder = require('../../../encoder')
const { ApiVersions: apiKey } = require('../../apiKeys')

  apiKey,
  apiVersion: 3,
  apiName: 'ApiVersions',
  encode: async () => {

})

  direction: 'request',
  fields: [
